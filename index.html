BEGIN; ALTER TABLE public.fire_events ADD COLUMN IF NOT EXISTS event_number bigint GENERATED ALWAYS AS IDENTITY (START WITH 100001); CREATE UNIQUE INDEX IF NOT EXISTS fire_events_event_number_unique ON public.fire_events (event_number); CREATE OR REPLACE FUNCTION public.keep_fire_event_number() RETURNS trigger LANGUAGE plpgsql SET search_path = '' AS $$ BEGIN IF NEW.event_number IS DISTINCT FROM OLD.event_number THEN RAISE EXCEPTION 'Event number cannot be changed'; END IF; RETURN NEW; END; $$; DROP TRIGGER IF EXISTS keep_fire_event_number ON public.fire_events; CREATE TRIGGER keep_fire_event_number BEFORE UPDATE OF event_number ON public.fire_events FOR EACH ROW EXECUTE FUNCTION public.keep_fire_event_number(); CREATE OR REPLACE FUNCTION public.worker_event_by_number( p_event_number text ) RETURNS TABLE (id uuid) LANGUAGE sql STABLE SECURITY DEFINER SET search_path = '' AS $$ SELECT e.id FROM public.fire_events e WHERE e.event_number::text = p_event_number AND e.status = 'active' LIMIT 1; $$; REVOKE ALL ON FUNCTION public.worker_event_by_number(text) FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.worker_event_by_number(text) TO anon, authenticated; COMMIT;
